⚡ ESP32 Cardboard Gun Arduino Code

Connect your cardboard gun's trigger push-button between GPIO 4 and GND on your ESP32. Flash this sketch using Arduino IDE at 115200 baud: 

// ============================================================
// TECHNITIX ESP32 CARDBOARD GUN TRIGGER (BAUD RATE: 115200)
// Connect Push-Button Trigger between GPIO 4 and GND
// ============================================================
const int TRIGGER_PIN = 4;
int lastState = HIGH;
unsigned long lastDebounce = 0;

void setup() {
  Serial.begin(115200);
  pinMode(TRIGGER_PIN, INPUT_PULLUP);
}

void loop() {
  int reading = digitalRead(TRIGGER_PIN);
  if (reading != lastState) {
    lastDebounce = millis();
  }
  
  if ((millis() - lastDebounce) > 40) {
    if (reading == LOW && lastState == HIGH) {
      // Trigger pulled! Sends SHOOT command to Chrome/Edge WebSerial
      Serial.println("SHOOT");
    }
  }
  lastState = reading;
}
